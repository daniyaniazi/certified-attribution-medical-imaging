from src.certify.smoothing import RandomizedSmoothingAttributor
from src.xai.attribution_unified import GradCAMUnified
from src.models.factory import get_model

model, _ = get_model('resnet18', num_classes=2)
smoother = RandomizedSmoothingAttributor(model, lambda img,cls: 
           GradCAMUnified(model).attribute(img, target_class=cls))
results = smoother.certify(image, k_percent=30, sigma=0.15, 
                           tau=0.75, num_samples=100)

# Results contain:
#   certified_map {-1,0,1}           (Eq. 5)
#   p_1, p_0 probabilities          (Eq. 5)  
#   certified_radius (Eq. 7)

═══════════════════════════════════════════════════════════════════════════════
KEY HYPERPARAMETERS (PAPER DEFAULTS)
═══════════════════════════════════════════════════════════════════════════════

σ (sigma) = 0.15
  Gaussian noise standard deviation

τ (tau) = 0.75
  Certification threshold (0.5 to <1.0)

K (k_percent) = [50, 30, 10]
  Sparsification level (keep top K%)

n (num_samples) = 100
  Number of smoothing samples

All customizable via smoother.certify() parameters
