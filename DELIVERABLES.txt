from src.certify.smoothing import RandomizedSmoothingAttributor
from src.xai.attribution_unified import GradCAMUnified
from src.models.factory import get_model

model, _ = get_model('resnet18', num_classes=2)
smoother = RandomizedSmoothingAttributor(model, 
           lambda img, cls: GradCAMUnified(model).attribute(img, cls))
results = smoother.certify(image, k_percent=30, sigma=0.15, tau=0.75, num_samples=100)

# Results: certified_map {-1,0,1}, p_1, p_0, certified_radius

═══════════════════════════════════════════════════════════════════════════════
KEY PARAMETERS (PAPER DEFAULTS)
═══════════════════════════════════════════════════════════════════════════════

σ (sigma) = 0.15
  Gaussian noise standard deviation

τ (tau) = 0.75
  Certification threshold

K (k_percent) = [50, 30, 10]
  Sparsification level

n (num_samples) = 100
  Number of smoothing samples

═══════════════════════════════════════════════════════════════════════════════
OUTPUT DESCRIPTION
═══════════════════════════════════════════════════════════════════════════════

results['certified_map']:
  Shape: [H, W]
  Values: {-1 (abstain), 0 (unimportant), 1 (important)}
  Color interpretation: Gray = uncertain, Red = unimportant, Green = important

results['p_1'], results['p_0']:
  Shape: [H, W]
  Range: [0, 1]
  Interpretation: Probability estimates for each pixel

results['certified_radius']:
  Single float value
  Formula: R = σ · Φ^(-1)(τ)
  Interpretation: Robustness guarantee for L2 perturbations

results['pct_certified']:
  Percentage of non-abstained pixels

results['pct_abstained']:
  Percentage of uncertain pixels
